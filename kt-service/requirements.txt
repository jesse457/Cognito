# Cognita Knowledge Tracing Service — Python dependencies
#
# NOTE ON pyBKT VERSION (deviation, see README.md "Deviation from spec"):
#   `pyBKT==1.0.*` installs on Python 3.12 but that release's sdist ships NO
#   `pyBKT/models/` package, so `from pyBKT.models import Model` fails. The
#   nearest WORKING pin is 1.4.3 (a plain wheel, no C++ compile needed).
pyBKT==1.4.3

fastapi
uvicorn[standard]
pydantic

# pyBKT 1.4.3's EM fitter relies on a chained ndarray assignment that raises
# "ValueError: setting an array element with a sequence" under NumPy >= 2.
# NumPy 1.26 is the last line that still permits it (with a DeprecationWarning).
numpy==1.26.4
pandas

# pyBKT hard-requires scikit-learn at runtime and its import-time metric probe
# crashes on scikit-learn >= 1.6 (it passes `list`s to functions expecting
# arrays). 1.5.2 is the newest version that imports cleanly.
# NOTE: pyBKT's setup.py also lists the deprecated `sklearn` shim as a build
# requirement, so a source build needs
#   SKLEARN_ALLOW_DEPRECATED_SKLEARN_PACKAGE_INSTALL=True
scikit-learn==1.5.2
scipy<1.14

# Test-only
pytest
httpx
